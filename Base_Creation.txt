USE [master];

CREATE DATABASE Perruque;
GO

USE Perruque;
GO

/***** Création de la TABLE TablePerruque *****/
CREATE TABLE [dbo].[TablePerruque](
	[IdPerruque] [int] IDENTITY(1,1) NOT NULL, 
	[Code] [varchar](4) NOT NULL
	
	
	/***** Ajout d'une CLÉ PRIMAIRE *****/
	CONSTRAINT [PK_Perruque] PRIMARY KEY ([IdPeruque]),

	/***** Ajout d'une CLÉ ÉTRANGÈRE *****/
	CONSTRAINT [FK_Perruque_Description] FOREIGN KEY ([IdDescription]) REFERENCES [dbo].[Description] (IdDescription),
	CONSTRAINT [FK_Perruque_Fournisseur] FOREIGN KEY ([IdFournisseur]) REFERENCES [dbo].[Fournisseur] (IdFournisseur)
)
GO

/***** Création de la TABLE Description *****/
CREATE TABLE [dbo].[Description](
	[IdDescription] [int] IDENTITY(1,1) NOT NULL, 
	[Style] [varchar](4) NOT NULL,
	[Coupe] [varchar](50) NOT NULL,
	[Couleur] [varchar](4) NOT NULL,
	[Longueur] [varchar](4) NOT NULL,
	[Genre] [varchar](4) NOT NULL,
	[Prix] [int] NOT NULL,
	[Qstock] [int] NOT NULL,
	[Qreserve] [int] NOT NULL

	/***** Ajout d'une CLÉ PRIMAIRE *****/
	CONSTRAINT [PK_Description] PRIMARY KEY ([IdDescription])
)
GO


/***** Création de la TABLE Fournisseur *****/
CREATE TABLE [dbo].[Fournisseur](
	[IdFournisseur] [int] IDENTITY(1,1) NOT NULL, 	
	[telephone] [varchar](6) NOT NULL,
	[IdSegment] [int] NOT NULL,
	[Nom] [varchar](50) NOT NULL,
	[Courriel] [varchar](50) NOT NULL
	
	/***** Ajout d'une CLÉ PRIMAIRE *****/
	CONSTRAINT [PK_Fournisseur] PRIMARY KEY ([IdFournisseur]),

	/***** Ajout d'une CLÉ ÉTRANGÈRE *****/
	CONSTRAINT [FK_Fournisseur_Segment] FOREIGN KEY ([IdSegment]) REFERENCES [dbo].[Segment] (IdSegment)
)
GO

/***** Création de la TABLE Segment *****/
CREATE TABLE [dbo].[Segment](
	[IdSegment] [int] IDENTITY(1,1) NOT NULL, 
	[Nom] [varchar](4) NOT NULL
	
	
	/***** Ajout d'une CLÉ PRIMAIRE *****/
	CONSTRAINT [PK_Segment] PRIMARY KEY ([IdSegment])
	
)
GO
